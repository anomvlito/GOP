\section{Módulo 2: Planificación de Corto Plazo y MRP}

\subsection{De lo agregado a lo desagregado}

\begin{definicion}{La cadena de desagregación}
   Planif. Agregada $\rightarrow$ MPS $\rightarrow$ MRP $\rightarrow$ Programación diaria

   \textbf{MPS (Plan Maestro de Producción):} traduce el plan agregado en productos específicos, cantidades específicas, períodos específicos. Responde ``¿cuántas unidades del producto X produzco en la semana Y?''. No dice qué materiales se necesitan.
   
   \textbf{MRP (Material Requirements Planning):} explota la demanda hacia atrás en el tiempo, considerando los plazos de entrega, para saber cuándo lanzar órdenes. Desarrollado por J. Orlicky en IBM.

   En el corto plazo: períodos más cortos, productos a nivel individual, demanda menos incierta, y aparecen complejidades como los setups.
\end{definicion}

\begin{definicion}{Insumos que necesita el MRP}
\begin{enumerate}
  \item \textbf{BOM (Bill of Materials):} la ``receta'' y estructura jerárquica del producto.
  \item \textbf{Plazo de entrega ($L$, lead time):} tiempo entre emitir la orden y recibirla.
  \item \textbf{Inventario disponible ($OH$, on hand):} stock de cada componente al comienzo.
\end{enumerate}
\end{definicion}

\subsection{La mecánica de las tablas MRP}

\begin{formulas}{Filas de la tabla MRP y su lógica}
   Para cada ítem se llena una tabla por semana con:
   \begin{itemize}
     \item \textbf{GR (Requerimiento Bruto):} demanda proveniente del nivel superior.
     \item \textbf{OH (Inventario Disponible):} stock al cierre del período. $OH_t = OH_{t-1} + \text{Recepciones}_t - GR_t$.
     \item \textbf{NR (Requerimiento Neto):} $NR_t = GR_t - OH_{t-1} - \text{en tránsito}_t$ (sólo si es $> 0$).
     \item \textbf{POR (Liberación de Orden Planificada):} orden desfasada $L$ períodos hacia atrás.
   \end{itemize}
   \textbf{Regla de oro:} la orden del padre se convierte en GR del hijo, multiplicada por la cantidad de la receta, en la semana en que el padre la libera (POR), no en la que la recibe.
\end{formulas}

\begin{trampa}{El desfase del Lead Time es donde todos se equivocan}
   Si una parte tiene $L = 3$ y necesito recibir 80 unidades en la semana 4 (NR), debo liberar la orden (POR) en la semana 1. El GR del componente hijo aparece en la semana 1 (cuando el padre libera), no en la semana 4. Confundir ``recepción'' con ``liberación'' arrastra el error a todo el árbol BOM.
\end{trampa}

\subsection{Lotificación: cómo decidir el tamaño del lote}

\begin{definicion}{Las técnicas de lotificación}
   Todas equilibran costo de ordenar/setup vs. costo de mantener inventario:
   \begin{itemize}
     \item \textbf{Lote a Lote (L4L):} produces exactamente lo necesario cada período. Cero inventario, ideal para ítems caros/perecederos; pero muchos setups.
     \item \textbf{Cantidad Fija de Pedido:} siempre la misma cantidad. Simple, pero genera inventario innecesario o insuficiente.
     \item \textbf{Período Fijo:} ordenar cada N períodos. Simple, pero inventario variable.
     \item \textbf{EOQ (cantidad económica de pedido):} lote fijo óptimo bajo demanda constante; rara en MRP.
     \item \textbf{Costo Total Mínimo:} ajusta el lote buscando igualar el costo de ordenar con el de inventario acumulado.
     \item \textbf{Costo Unitario Mínimo:} divide el costo total entre las unidades del lote y minimiza ese cociente.
     \item \textbf{Silver-Meal:} heurística que minimiza el costo promedio por período.
     \item \textbf{Wagner-Whitin (WW):} el óptimo matemático exacto (prog. dinámica), pero costoso y poco intuitivo.
   \end{itemize}
\end{definicion}

\begin{formulas}{Algoritmo Silver-Meal (paso a paso)}
   Se busca el lote que minimiza el costo promedio por período cubierto. Para un lote que arranca en el período 1 y cubre hasta el período $k$:
   $$CT(k) = \frac{S + \sum_{t=1}^{k} h \cdot (t - 1) \cdot D_t}{k}$$
   donde $S$ = costo de setup, $h$ = costo de mantener por unidad-período, $D_t$ = demanda del período $t$. 
   \textbf{Regla de parada:} se sigue agregando períodos al lote mientras $CT(k)$ decrezca; en cuanto $CT(k) > CT(k - 1)$, se cierra el lote en $k - 1$ y se reinicia el conteo desde $k$.
\end{formulas}

\begin{trampa}{Silver-Meal: no empieces en períodos con demanda cero}
   Si los primeros períodos tienen demanda = 0, sáltate directamente al primer período con demanda positiva y arranca Silver-Meal desde ahí. $Q = 0$ en los períodos con demanda cero es trivial: no hay nada que ordenar. Empezar ahí genera comparaciones degeneradas ($CT(1) = 0$) que confunden sin aportar información.
\end{trampa}

\begin{tipprueba}{¿Cuándo usar cada método en la prueba?}
   \textbf{Identificación:} si el enunciado dice ``use L4L'' $\implies$ produces exacto, $OH$ siempre 0. Si dice ``Silver-Meal'' $\implies$ arma una tabla de costos promedio con regla de parada. Si dice ``Wagner-Whitin'' $\implies$ matriz de horizonte (último período de producción vs. horizonte) y se elige el mínimo $Z_t$. 
   \textbf{Qué aplicar:} en Silver-Meal arma una mini-tabla de $CT(k)$ por cada lote candidato; el inventario se valoriza por las semanas que ``espera'' ($(t - 1) \text{ períodos} \times h$).
\end{tipprueba}

\subsection{Comparación de los sistemas (ejemplo del curso)}

\begin{definicion}{Resultados comparativos (ejemplo de clase, 8 semanas)}
   Con $S = \$47$, costo artículo $\$10$, $h = 0,5\%$ del valor:
   
   \begin{center}
   \begin{tabular}{lc}
     \toprule
     \textbf{Método} & \textbf{Costo Total} \\
     \midrule
     Lote a Lote (L4L) & $\$376,00$ \\
     EOQ de Wilson & $\$171,05$ \\
     Costo Unitario Mínimo & $\$153,50$ \\
     Costo Total Mínimo & $\$140,05$ \\
     \bottomrule
   \end{tabular}
   \end{center}

   \textbf{Conclusión clave:} producir por lotes (no ajustarse a cada pedido) reduce costos. Por eso las empresas producen por lotes y no L4L puro.
\end{definicion}

\subsection{MRP de ciclo cerrado y métricas TPM/OEE}

\begin{definicion}{MRP de ciclo cerrado y OEE}
   El MRP de ciclo cerrado retroalimenta la planeación de capacidad: si el plan no es factible, se replantea el MPS y se repite el proceso período a período. En mantenimiento, el TPM (Mantenimiento Productivo Total) mide el rendimiento de los equipos mediante la OEE (Eficiencia Global del Equipo):
   $$OEE = \text{Disponibilidad} \times \text{Rendimiento} \times \text{Calidad} = \frac{B}{A} \times \frac{C}{B} \times \frac{D}{C} = \frac{D}{A}$$
   donde $A$=tiempo planificado, $B$=tiempo operativo, $C$=tiempo de funcionamiento, $D$=tiempo productivo.
\end{definicion}

\subsection{V/F frecuentes de Planificación y MRP}
\begin{enumerate}[label=\textbf{\arabic*.}]
  \item \textbf{Falso:} ``El MPS indica qué materiales y en qué cantidad comprar.'' — Falso: eso lo hace el MRP; el MPS solo dice cuántos productos terminados producir.
  \item \textbf{Verdadero:} ``El MRP explota la demanda hacia atrás considerando los lead times.''
  \item \textbf{Falso:} ``L4L minimiza el número de setups.'' — Falso: L4L minimiza el inventario pero suele maximizar los setups.
  \item \textbf{Verdadero:} ``Wagner-Whitin entrega la solución óptima exacta de lotificación.''
  \item \textbf{Falso:} ``Agregar la demanda aumenta el error de pronóstico.'' — Falso: la agregación reduce el error relativo (los errores se compensan).
  \item \textbf{Verdadero:} ``La estrategia Chase minimiza el inventario pero eleva los costos de contratación/despido.''
\end{enumerate}

\subsection{MRP — BOM, L4L y Silver-Meal (I2 2025, Pregunta 3)}

\begin{tipprueba}{Ensamble con BOM, L4L y Silver-Meal (I2 2025 / Ay. 8)}
   \textbf{Identificación:} producto final con BOM, lead times, inventario inicial y orden en tránsito. 
   \textbf{Qué aplicar:} (1) dibujar BOM, (2) tabla MRP del producto final con L4L (desfasar por $L$), (3) Silver-Meal para el componente hijo.

   \textbf{i) Árbol BOM:}
   \begin{verbatim}
          Producto Final
         /       |       \
    A (1)      B (2)      C (1)
   /    \     /    \
A1(1) A2(1) A1(2) B2(1)
   \end{verbatim}

   \textbf{ii) MRP del producto final (L4L).} Demanda: sem 4=10, 5=50, 6=40, 7=60, 8=50. $OH_0 = 50$; orden en tránsito 20 que llega en sem 2; $L = 1$ (ensamble).

   \begin{center}
   \small
   \begin{tabular}{lccccccccc}
     \toprule
     \textbf{Período} & \textbf{0} & \textbf{1} & \textbf{2} & \textbf{3} & \textbf{4} & \textbf{5} & \textbf{6} & \textbf{7} & \textbf{8} \\
     \midrule
     Requerimiento Bruto (GR) & & 0 & 0 & 0 & 10 & 50 & 40 & 60 & 50 \\
     Prod. Tránsito & & 0 & 20 & 0 & 0 & 0 & 0 & 0 & 0 \\
     Inventario Final (OH) & 50 & 50 & 70 & 70 & 60 & 10 & 0 & 0 & 0 \\
     Requerimiento Neto (NR) & & 0 & 0 & 0 & 0 & 0 & 30 & 60 & 50 \\
     Orden Programada (POR) & & 0 & 0 & 0 & 0 & 30 & 60 & 50 & 0 \\
     \bottomrule
   \end{tabular}
   \end{center}

   Con L4L se produce exactamente el requerimiento neto, desfasado 1 semana.
\end{tipprueba}

\begin{tipprueba}{¿Cómo se calcula la demanda de un hijo? (Conexión clave)}
   \textbf{Regla general de explosión:} La demanda bruta (GR) de un componente hijo equivale a la suma de las liberaciones de órdenes (POR) de todos sus padres directos en la misma semana, multiplicadas por el factor de uso del árbol BOM:
   $$GR_{\text{hijo},t} = \sum_{\text{padres}} (POR_{\text{padre},t} \times \text{factor})$$
   
   Sumando ambos padres, la demanda teórica de A1 sería (como ejemplo ilustrativo):
   \begin{itemize}
     \item Semana 2: 30 (dada en el enunciado por necesidades iniciales/en tránsito).
     \item Semana 3: $POR_{A,3} + POR_{B,3} + \text{demanda base} = 30 + 60 + 30 = 120$.
     \item Semana 4: $POR_{A,4} + POR_{B,4} = 60 + 120 = 180$.
     \item Semana 5: $POR_{A,5} + POR_{B,5} = 50 + 100 = 150$.
   \end{itemize}

   \textbf{iii) Sabemos que el requerimiento neto de partes A1 es:}
   \begin{center}
   \begin{tabular}{lcccccccc}
     \toprule
     \textbf{Período} & \textbf{1} & \textbf{2} & \textbf{3} & \textbf{4} & \textbf{5} & \textbf{6} & \textbf{7} & \textbf{8} \\
     \midrule
     \textbf{Demanda} & 0 & 30 & 120 & 180 & 150 & 0 & 0 & 0 \\
     \bottomrule
   \end{tabular}
   \end{center}

   En la \textbf{primera iteración} (en $t = 1$ técnico, semana 2 real) de S-M:
   \begin{itemize}
     \item $k = 1$: $CT(1) = 120$.
     \item $k = 2$: $CT(2) = \frac{120 + (120 \times 0.9)}{2} = 114 < CT(1)$, así que seguimos.
     \item $k = 3$: $CT(3) = \frac{120 + (120 \times 0.9) + (180 \times 2 \times 0.9)}{3} = 184 > CT(2)$.
   \end{itemize}
   Paramos y agrupamos $k=2$ períodos. Lote $Q_1 = 30 + 120 = 150$.

   \textbf{Conclusión S-M:} Agrupar las semanas ahorra setups, pero las semanas 4 y 5 conviene producirlas solas debido a su alto volumen de demanda, el cual haría muy costoso mantenerlas en inventario.
\end{tipprueba}
